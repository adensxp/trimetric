## 🔁 执行步骤

### 1️⃣ 读取三套预备池
```python
import sqlite3
from datetime import datetime

today = datetime.now().strftime('%Y-%m-%d')
conn = sqlite3.connect('/workspace/stock_db/candidates.db')

for strategy, label in [('SHORT_TERM', '短线'), ('LONG_TERM', '长线'), ('CORE_HOLDING', '底仓')]:
    cur = conn.execute('''
        SELECT rank, code, name, total_score, signal_count
        FROM daily_prep_pool
        WHERE prep_date = ? AND strategy = ?
        ORDER BY rank
    ''', (today, strategy))
    pool = cur.fetchall()
    print(f'{label}预备池:')
    for r in pool:
        print(f'  #{r[0]} {r[1]} {r[2]}: 评分 {r[3]} 信号 {r[4]}')
conn.close()
```
