### 6️⃣ 【关键】持仓规则实时监控

```python
import sqlite3
conn = sqlite3.connect('/workspace/stock_db/positions.db')
positions = conn.execute("SELECT code, name, shares, cost_basis, entry_price FROM positions WHERE active=1").fetchall()
for code, name, shares, cost, entry in positions:
    cur = conn.execute("SELECT close_price FROM stock_daily WHERE stock_code = ? ORDER BY trade_date DESC LIMIT 1", (code,))
    row = cur.fetchone()
    if row:
        price = row[0]
        pnl_pct = (price / entry - 1) * 100
        pnl_amount = (price - entry) * shares

        alerts = []
        if pnl_pct <= -5:
            alerts.append(f'🔴 R005 强平：当前 {price} ≤ 止损 {entry*0.95:.2f} 浮亏 {pnl_pct:.2f}%')
        elif pnl_pct < 0:
            alerts.append(f'🔴 R024 红线：浮盈变浮亏 {pnl_pct:.2f}%')
        elif pnl_pct >= 10 and price >= entry * 1.10:
            alerts.append(f'🟢 R008 加仓 2：浮盈 +{pnl_pct:.2f}%，价格 {price} ≥ {entry*1.10:.2f}')
        elif pnl_pct >= 5 and price >= entry * 1.05:
            alerts.append(f'🟢 R007 加仓 1：浮盈 +{pnl_pct:.2f}%，价格 {price} ≥ {entry*1.05:.2f}')

        print(f"=== {code} {name} ===")
        print(f"  成本 {entry:.2f} → 现价 {price:.2f} 浮盈 {pnl_amount:+,.0f} ({pnl_pct:+.2f}%)")
        if alerts:
            for a in alerts:
                print(f"  {a}")
        else:
            print(f"  ✅ 无触发")
```
